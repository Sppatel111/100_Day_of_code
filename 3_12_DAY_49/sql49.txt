# sql aliases

#as keyword
select customer_id as id from Customers;

#WE CAN ALSO USE WITHOUT AS KEYWORD
select customer_id id from Customers;

select customer_id AS  id,first_name from Customers;

select customer_id AS  id,first_name as [my grerat customer] from Customers;

#cloumn concatination
select customer_id as  id ,(first_name ||' '|| last_name) as name from Customers;

#not work in online editor
select customer_id as  id , first_name +' '+ last_name  as name from Customers;
#for mysql
select customer_id as  id ,CONCAT(first_name +' '+ last_name) as name from Customers;

#table as different name
#for the table it useless but forte more table queries you perform in one query than it is usefull

select * from Customers as Students;

select c.customer_id,o.order_id, c.first_name

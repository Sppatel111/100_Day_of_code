CREATE OR REPLACE PROCEDURE add_user(e VARCHAR, p VARCHAR, d VARCHAR)
LANGUAGE plpgsql
AS $
BEGIN
    INSERT INTO user_auth(email, password, department) VALUES (e, p, d);
END;$;

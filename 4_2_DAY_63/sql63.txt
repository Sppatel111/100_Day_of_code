##joins
##INNER JOIN
 select empdetails.empname,empdetails.empid,empdept.deptname,empdept.dept_off
 from empdetails inner join empdept on  empdetails.department=empdept.deptid;


## LEFT JOIN

select empdetails.empname,empdetails.empid,empdept.deptname,empdept.dept_off from
empdetails left join empdept on  empdetails.department=empdept.deptid;

## RIGHT JOIN

 select empdetails.empname,empdetails.empid,empdept.deptname from empdetails
 right join empdept on  empdetails.department=empdept.deptid;


## FULL OUTER JOIN

select empdetails.empname,empdetails.empid,empdept.deptname,empdept.dept_off from
empdetails full outer  join empdept on  empdetails.department=empdept.deptid;


## SELF JOIN

select A.empname as e1,B.empname as e2 from empdetails A, empdetails B where A.empid <> B.empid ;

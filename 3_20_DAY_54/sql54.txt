## logical operators

## ANY and ALL use for the comparison( = < > <>)

ALL: all the subquery meets the condition

SELECT ProductName
FROM Products
WHERE ProductID = ALL (SELECT ProductID FROM OrderDetails WHERE Quantity = 10);

ANY: if any subquery meets the condition  --or--
SOME:if any of the subquery value meets the condition.

SOME AND OR EQUIVALENT

select * from Orders
where CustomerID = ANY(select CustomerID from Customers where Country = 'Germany' );


AND:if both condition need to be  true

select * from Orders where customer_id =4 and amount=400;

BETWEEN: it show between the condition

select * from Orders where  amount between 200 and 400;

EXISTS:
true when subquery return one or more records
select * from Customers
 where exists(select item from Orders where Customers.customer_id=Orders.customer_id);

IN:
return true if one of the records are in match.

select * from Customers where country IN('USA','UAE');

select * from Customers where first_name like 'J%';

select * from Customers where first_name not like 'J%';
